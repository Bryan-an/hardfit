> ### Locked technical facts (verified 2026-06-01 via the M2 workflow — do not re-derive)
> **Anderson-Darling A² (raw):** sorted `u_i = clamp(F(x_(i)), [1e-12, 1−1e-12])`; `A² = −n − (1/n)·Σ_{i=1..n}(2i−1)[ln u_i + ln(1 − u_{n+1−i})]` (0-based: weight `2i+1`, reversed index `u[n−1−i]`). Clamp negative FP noise to 0. **Use the CANONICAL route:** modify the statistic UP to `A*²` and compare to FIXED published critical values — do NOT also divide criticals (that is scipy's alternative; mixing them double-corrects).
> **AD p-value taxonomy:** Normal/Lognormal → closed-form `A*² = A²(1 + 0.75/n + 2.25/n²)` + D'Agostino-Stephens piecewise formula (lognormal = normal test on ln x; A² is log-invariant so compute from the lognormal CDF on raw x). Exponential → `A*² = A²(1 + 0.6/n)`, bracketed p via the table `[0.916,1.062,1.321,1.591,1.959]` at `[15,10,5,2.5,1]%`. **Weibull (2-param) → `A*² = A²(1 + 0.2/√n)`, FIXED Gumbel/EV table `[0.474,0.637,0.757,0.877,1.038]` at `[25,10,5,2.5,1]%`** (NOT scipy's shape-dependent table — that's for 3-param Weibull). Gamma → no closed form; `method: 'diagnostic'` (bootstrap p-value comes in M2.2).
> **Cramér-von Mises:** `W² = 1/(12n) + Σ(u_i − (2i−1)/(2n))²` (scipy's `n·ω²` normalization — match it); modified `W*² = W²(1 + 0.5/n)`. p-value: diagnostic for now (bootstrap in M2.2; optional normal-only nortest polynomial).
> **Chi-Squared (continuous):** equiprobable bins via the FITTED quantile: edges `Q(j/k)` for `j=1..k−1`, `k = max(2, ⌊n/5⌋)` so `E_j = n/k ≥ 5`; `X² = Σ(O_j − E_j)²/E_j`; `df = k − 1 − p`; p = `1 − chisquareCdf(X², df)` (`@stdlib/stats-base-dists-chisquare-cdf@0.3.1`, `cdf(x, df)`). Quantile-based bins handle bounded support automatically. **Chernoff-Lehmann caveat:** `χ²(k−1−p)` is anti-conservative for raw-data MLE → label the p-value approximate (mirror the KS Lilliefors note); the rigorous p-value is the bootstrap (M2.2).
> **Quantile packages (CommonJS export=, pin individually):** `@stdlib/stats-base-dists-{normal,exponential,weibull}-quantile@0.3.1`, `-lognormal-quantile@0.2.3`, `-gamma-quantile@0.2.3`. **Conventions match the M1 CDFs:** gamma `quantile(p, shape, RATE)` (pass `p.rate`, the same slot-trap as gamma cdf — add a convention-guard test), weibull `quantile(p, shape, SCALE)`, exponential `quantile(p, RATE)`, normal/lognormal `quantile(p, mu, sigma)`.
> **scipy↔HardFit parameter mapping (the fixture gate's load-bearing detail):** `expon.fit(x, floc=0)`→`(0,scale)`, HardFit `rate=1/scale`. `gamma.fit(x, floc=0)`→`(a,0,scale)`, HardFit `shape=a, rate=1/scale`. `weibull_min.fit(x, floc=0)`→`(c,0,scale)`, HardFit `shape=c, scale=scale`. `lognorm.fit(x, floc=0)`→`(s,0,scale)`, HardFit `mu=ln(scale), sigma=s`. `norm.fit(x)`→`(loc,scale)` (NO floc), HardFit `mu=loc, sigma=scale` (scipy scale = ÷n population std = HardFit's). **Fixture gate:** Mode B = GoF arithmetic at FIXED scipy-derived params, `rtol 1e-9`. export function adStatistic(data: readonly number[], cdf: (x: number) => number): number {
  const x = [...data].sort((a, b) => a - b)
  const n = x.length
  if (n === 0) return Number.NaN
  const EPS = 1e-12
  const u = x.map((xi) => Math.min(1 - EPS, Math.max(EPS, cdf(xi))))
  let s = 0
  for (let i = 0; i < n; i++) {
    const w = 2 * i + 1 // (2·rank − 1), rank = i+1
    const lo = u[i] as number // F(x_(i))
    const hi = u[n - 1 - i] as number // F(x_(n+1−i)) — reversed index
    s += w * (Math.log(lo) + Math.log(1 - hi))
  }
  return Math.max(0, -n - s / n)
}
```

export function cramerVonMises(data: readonly number[], cdf: (x: number) => number): number {
  const x = [...data].sort((a, b) => a - b)
  const n = x.length
  if (n === 0) return Number.NaN
  let sum = 0
  for (let i = 0; i < n; i++) {
    const u = Math.min(1, Math.max(0, cdf(x[i] as number)))
    const d = u - (2 * (i + 1) - 1) / (2 * n)
    sum += d * d
  }
  return 1 / (12 * n) + sum
}
```

import chi2cdf from '@stdlib/stats-base-dists-chisquare-cdf'

const MIN_EXPECTED_PER_BIN = 5

/** Chi-squared GoF for a CONTINUOUS fit using equiprobable bins from the fitted quantile. */
export function chiSquaredGof(
  data: readonly number[],
  quantile: (prob: number) => number,
  nParams: number,
): { statistic: number; df: number; bins: number; pValue: number } {
  const n = data.length
  const k = Math.max(2, Math.floor(n / MIN_EXPECTED_PER_BIN))
  const edges: number[] = []
  for (let j = 1; j < k; j++) edges.push(quantile(j / k))
  const observed = new Array<number>(k).fill(0)
  for (const x of data) {
    let b = 0
    while (b < edges.length && x > (edges[b] as number)) b++
    observed[b] = (observed[b] as number) + 1
  }
  const expected = n / k
  let statistic = 0
  for (let j = 0; j < k; j++) {
    const d = (observed[j] as number) - expected
    statistic += (d * d) / expected
  }
  const df = k - 1 - nParams
  const pValue = df >= 1 ? 1 - chi2cdf(statistic, df) : Number.NaN
  return { statistic, df, bins: k, pValue }
}
```

import { DistributionName } from './types'
import type { GofResult, PValueMethod } from './types'

// D'Agostino & Stephens (1986) piecewise p-value for the MODIFIED normal A*².
export function adNormalPValue(aStar: number): number {
  const a = aStar
  if (a >= 0.6) return Math.exp(1.2937 - 5.709 * a + 0.0186 * a * a)
  if (a > 0.34) return Math.exp(0.9177 - 4.279 * a - 1.38 * a * a)
  if (a > 0.2) return 1 - Math.exp(-8.318 + 42.796 * a - 59.938 * a * a)
  return 1 - Math.exp(-13.436 + 101.14 * a - 223.73 * a * a)
}

// Stephens critical-value tables (A*²). Significance levels paired with criticals.
const EXP_SIG = [0.15, 0.1, 0.05, 0.025, 0.01]
const EXP_CRIT = [0.916, 1.062, 1.321, 1.591, 1.959] // exponential
const GUMBEL_SIG = [0.25, 0.1, 0.05, 0.025, 0.01]
const GUMBEL_CRIT = [0.474, 0.637, 0.757, 0.877, 1.038] // 2-param Weibull via Gumbel/EV

/** Bracketed p by linear interpolation of A*² in a critical-value table. */
function bracketedPValue(aStar: number, sig: number[], crit: number[]): number { /* interpolate; clamp to (min,max) ends */ }

/** Returns the AD GofResult for a distribution given the RAW A² and n. */
export function adResult(name: string, a2: number, n: number): GofResult {
  switch (name) {
    case DistributionName.Normal:
    case DistributionName.Lognormal: {
      const aStar = a2 * (1 + 0.75 / n + 2.25 / (n * n))
      return { statistic: a2, pValue: adNormalPValue(aStar), method: 'closed-form' satisfies PValueMethod }
    }
    case DistributionName.Exponential: {
      const aStar = a2 * (1 + 0.6 / n)
      return { statistic: a2, pValue: bracketedPValue(aStar, EXP_SIG, EXP_CRIT), method: 'table' }
    }
    case DistributionName.Weibull: {
      const aStar = a2 * (1 + 0.2 / Math.sqrt(n))
      return { statistic: a2, pValue: bracketedPValue(aStar, GUMBEL_SIG, GUMBEL_CRIT), method: 'table' }
    }
    default: // gamma + anything else: no closed-form AD critical values
      return { statistic: a2, pValue: null, method: 'diagnostic' }
  }
}
```
